# Home-screen WidgetKit widget Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** A home-screen iOS widget (small + medium) that mirrors the in-app Teraz hero for the active place, by having the RN app bake a versioned snapshot into an App Group that the widget draws verbatim.

**Architecture:** App computes everything in TS (`buildWidgetSnapshot`, same `scene`/`displayValue`/`formatPollutant` as the app) → publishes via a `WidgetSync` seam → native module writes App-Group UserDefaults + reloads the widget. Tasks 1–3 are headless-testable TS; Task 4 is the interactive native gate (Swift/Xcode) done with the human; Task 5 is manual sim verification.

**Tech Stack:** TypeScript strict, React Native 0.86, Jest + @testing-library/react-native (render/renderHook are async — await them). WidgetKit/SwiftUI + a custom `RCTBridgeModule` (Task 4). No new npm dependency.

## Global Constraints
- No new npm dependency (WidgetKit/App Group/native module are native, not packages). · TS strict, no `any` outside a justified data boundary. · `src/core` pure (zero React/data imports), enforced 100% coverage (jest.config `./src/core/`). · Files ≤200 lines, functions ≤40. · Imports one-way: features→shared→core, data→core; features never import data. · Behavior tests cite AC IDs.
- Spec `docs/specs/017-widget.md` is authoritative (exact snapshot fields, AC values). Snapshot is **verbatim-draw** — the widget recomputes nothing.
- App Group suite string (Task 4): `group.org.reactjs.native.example.Powietrze`.

```ts
import type { Reading, ReadingDetail } from '../air';
import { displayValue, scaleLabel, formatPollutant, POLLUTANTS } from '../air';
import { scene, bandOf, BANDS } from '../scene';
import type { Scale, Precision } from '../settings';

export const WIDGET_SNAPSHOT_VERSION = 1;
export interface WidgetTile { label: string; value: string; unit: string }
export interface WidgetSnapshot {
  version: number; city: string; stationLabel: string;
  displayValue: string; scaleCaption: string; band: string;
  keyHex: string; deepHex: string; midHex: string;
  tiles: WidgetTile[]; measuredAt: string;
}
export interface WidgetSync { publish(snapshot: WidgetSnapshot): void }

export function buildWidgetSnapshot(
  reading: Reading, detail: ReadingDetail | undefined,
  scale: Scale, precision: Precision,
): WidgetSnapshot {
  const s = scene(reading.index);
  const tiles = (detail?.pollutants ?? []).map(p => ({
    label: POLLUTANTS.find(c => c.code === p.code)?.label ?? p.code,
    value: formatPollutant(p.value, precision),
    unit: 'µg/m³',
  }));
  return {
    version: WIDGET_SNAPSHOT_VERSION,
    city: reading.city, stationLabel: reading.station,
    displayValue: displayValue(reading.index, reading.pm25, scale, precision),
    scaleCaption: scaleLabel(scale),
    band: BANDS[bandOf(reading.index)],
    keyHex: s.key, deepHex: s.deep, midHex: s.mid,
    tiles, measuredAt: reading.measuredAt,
  };
}

// Stable string that changes iff a republish is warranted (drives the provider's dedup).
export function widgetSnapshotIdentity(s: WidgetSnapshot): string {
  return [s.city, s.stationLabel, s.measuredAt, s.displayValue, s.band].join('|');
}
```

```tsx
import { useEffect, useRef, type ReactNode } from 'react';
import { useActivePlace } from '../place';
import { useSettings } from '../settings';
import { buildWidgetSnapshot, widgetSnapshotIdentity, type WidgetSync } from '../../core/widget';

export function WidgetSyncProvider({ sync, children }: { sync: WidgetSync; children: ReactNode }) {
  const { status, reading, detail } = useActivePlace();
  const { settings } = useSettings();
  const lastId = useRef<string | null>(null);
  useEffect(() => {
    if (status !== 'ready' || !reading) return;
    const snap = buildWidgetSnapshot(reading, detail, settings.scale, settings.precision);
    const id = widgetSnapshotIdentity(snap);
    if (id === lastId.current) return;
    lastId.current = id;
    sync.publish(snap);
  }, [status, reading, detail, settings.scale, settings.precision, sync]);
  return <>{children}</>;
}
```

```ts
import { NativeModules } from 'react-native';
import type { WidgetSync, WidgetSnapshot } from '../../core/widget';

export function createNativeWidgetSync(): WidgetSync {
  const mod = NativeModules.WidgetSync as
    | { writeSnapshot(json: string): void; reloadTimelines(): void }
    | undefined;
  return {
    publish(snapshot: WidgetSnapshot) {
      if (!mod) return; // Android / pre-extension build → no-op
      mod.writeSnapshot(JSON.stringify(snapshot));
      mod.reloadTimelines();
    },
  };
}
```
